    ```rust
    impl FlushClient {
        pub fn new(
            api_key: String,
            metadata_svc_url: String,
            org_id: String,
            timeout: Duration,
        ) -> Result<Self, reqwest::Error> {
            let client = reqwest::Client::builder()
                .timeout(timeout)
                .build()?;
            Ok(Self { client, api_key, metadata_svc_url, org_id })
        }

        pub async fn flush_pending(
            &self,
            pending: &HashMap<String, MetricTypeInfo>,
        ) -> Result<Vec<String>, FlushError> {
            let body = build_request_body(&self.org_id, pending);
            let url = format!(
                "{}/api/unstable/byoc/ingest/metadata/metric-metadata",
                self.metadata_svc_url
            );

            let response = self
                .client
                .post(&url)
                .header("DD-API-KEY", &self.api_key)
                .json(&body)
                .send()
                .await
                .map_err(|err| {
                    if err.is_timeout() {
                        FlushError::Timeout
                    } else {
                        FlushError::Network(err.to_string())
                    }
                })?;

            let status = response.status();
            if !status.is_success() {
                let body_text = response.text().await.unwrap_or_default();
                warn!(
                    status = status.as_u16(),
                    body = %body_text,
                    "metadata flush failed"
                );
                return Err(FlushError::HttpStatus {
                    status: status.as_u16(),
                    body: body_text,
                });
            }

            let api_response: UpsertResponse = response
                .json()
                .await
                .map_err(|err| FlushError::ResponseParse(err.to_string()))?;

            Ok(api_response.succeeded_metrics)
        }
    }

    fn build_request_body(
        org_id: &str,
        pending: &HashMap<String, MetricTypeInfo>,
    ) -> UpsertRequest {
        let records = pending
            .iter()
            .map(|(name, info)| UpsertRecord {
                metric_name: name.clone(),
                metric_type: info.metric_type,
                interval: i64::from(info.interval),
            })
            .collect();
        UpsertRequest {
            org_id: org_id.to_string(),
            records,
        }
    }
    ```

    Key implementation details:
    - `warn!` log on HTTP failure per D-08 -- message starts with lowercase, no trailing punctuation
    - `i64::from(info.interval)` casts u32 -> i64 for wire format compatibility with Go int64
    - `#[serde(skip_serializing_if = "is_zero")]` on interval field matches Go omitempty
    - `#[serde(default)]` on succeeded_metrics handles null, missing, and empty cases
    - No `unwrap()` in library code -- all errors propagated via `?` or `map_err`
    - No retry logic per D-08

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| FlushClient -> SaaS endpoint | Outbound HTTP POST crosses network boundary to external metadata service |
| DD-API-KEY in header | Secret credential sent over the wire on every flush |
| Response body parsing | Untrusted JSON response deserialized into UpsertResponse |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | S (Spoofing) | FlushClient::flush_pending | mitigate | DD-API-KEY header authenticates requests; key validated at build() (Phase 1 T-01-01). TLS via reqwest rustls-tls feature encrypts in transit. |
| T-03-02 | I (Info Disclosure) | FlushClient struct | mitigate | Debug NOT derived on FlushClient -- api_key field excluded from log output. FlushError does NOT include api_key in any variant's Display impl. warn! log on failure includes status and body but never api_key. |
| T-03-03 | T (Tampering) | UpsertResponse deserialization | accept | Response is read-only (we extract succeeded_metrics). Malformed JSON returns FlushError::ResponseParse, not a panic. Attacker controlling response can only cause metrics to not be marked as known (DoS, not data corruption). |
| T-03-04 | D (Denial of Service) | FlushClient timeout | mitigate | reqwest::Client built with configurable timeout (default 10s per CFG-03). Timeout returns FlushError::Timeout, caller drops pending list and continues. No retry loop that could amplify. |
| T-03-05 | E (Elevation) | N/A | accept | FlushClient runs in-process with same privileges as the transform. No privilege boundaries within the component. |
</threat_model>
