//! Query execution configuration for the metrics engine.

use serde::{Deserialize, Serialize};

/// Configuration for metrics query execution.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct MetricsQueryConfig {
    /// Maximum memory per query in bytes.
    /// Default: 512MB
    pub max_memory_bytes: usize,

    /// Target number of partitions for parallel execution.
    /// Default: number of CPU cores
    pub target_partitions: usize,

    /// Batch size for query execution.
    /// Default: 8192 rows
    pub batch_size: usize,

    /// Enable query result caching.
    /// Default: false (stateless queries preferred for metrics)
    pub enable_cache: bool,

    /// Parquet read options.
    pub parquet: ParquetReadConfig,
}

/// Parquet-specific read configuration.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ParquetReadConfig {
    /// Enable pushdown predicates.
    /// Default: true
    pub pushdown_filters: bool,

    /// Enable row group pruning.
    /// Default: true
    pub prune_row_groups: bool,

    /// Read page index for better pruning.
    /// Default: true
    pub page_index: bool,
}

impl Default for MetricsQueryConfig {
    fn default() -> Self {
        Self {
            max_memory_bytes: 512 * 1024 * 1024, // 512MB
            target_partitions: num_cpus::get(),
            batch_size: 8192,
            enable_cache: false,
            parquet: ParquetReadConfig::default(),
        }
    }
}

impl Default for ParquetReadConfig {
    fn default() -> Self {
        Self {
            pushdown_filters: true,
            prune_row_groups: true,
            page_index: true,
        }
    }
}

impl MetricsQueryConfig {
    /// Create config optimized for high-throughput ingestion queries.
    pub fn for_ingest() -> Self {
        Self {
            max_memory_bytes: 256 * 1024 * 1024, // 256MB - lighter for ingest
            batch_size: 16384, // Larger batches for throughput
            ..Default::default()
        }
    }

    /// Create config optimized for aggregation queries.
    pub fn for_aggregation() -> Self {
        Self {
            max_memory_bytes: 1024 * 1024 * 1024, // 1GB for aggregations
            batch_size: 8192,
            ..Default::default()
        }
    }
}

//! DataFusion SessionContext wrapper for metrics queries.

use std::sync::Arc;

use datafusion::execution::context::{SessionConfig, SessionContext, SessionState};
use datafusion::execution::runtime_env::{RuntimeConfig, RuntimeEnv};
use datafusion::prelude::*;

use super::config::MetricsQueryConfig;

/// Error types for query context operations.
#[derive(Debug, thiserror::Error)]
pub enum ContextError {
    #[error("DataFusion error: {0}")]
    DataFusion(#[from] datafusion::error::DataFusionError),

    #[error("Configuration error: {0}")]
    Config(String),
}

/// Metrics query session context.
///
/// Wraps DataFusion's SessionContext with metrics-specific configuration
/// and provides a clean interface for query execution.
pub struct MetricsSessionContext {
    inner: SessionContext,
    config: MetricsQueryConfig,
}

impl MetricsSessionContext {
    /// Create a new MetricsSessionContext with the given configuration.
    pub fn new(config: MetricsQueryConfig) -> Result<Self, ContextError> {
        let runtime_config = RuntimeConfig::new()
            .with_memory_limit(config.max_memory_bytes, 1.0);

        let runtime_env = Arc::new(RuntimeEnv::new(runtime_config)?);

        let session_config = SessionConfig::new()
            .with_target_partitions(config.target_partitions)
            .with_batch_size(config.batch_size)
            .with_parquet_pushdown_filters(config.parquet.pushdown_filters)
            .with_parquet_pruning(config.parquet.prune_row_groups);

        let state = SessionState::new_with_config_rt(session_config, runtime_env);
        let inner = SessionContext::new_with_state(state);

        Ok(Self { inner, config })
    }

    /// Create with default configuration.
    pub fn with_defaults() -> Result<Self, ContextError> {
        Self::new(MetricsQueryConfig::default())
    }

    /// Get the underlying DataFusion SessionContext.
    pub fn inner(&self) -> &SessionContext {
        &self.inner
    }

    /// Get the query configuration.
    pub fn config(&self) -> &MetricsQueryConfig {
        &self.config
    }

    /// Execute a SQL query against registered tables.
    pub async fn sql(&self, query: &str) -> Result<DataFrame, ContextError> {
        Ok(self.inner.sql(query).await?)
    }

    /// Register a Parquet file as a table.
    pub async fn register_parquet(
        &self,
        name: &str,
        path: &str,
    ) -> Result<(), ContextError> {
        self.inner.register_parquet(name, path, Default::default()).await?;
        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_context_creation() {
        let ctx = MetricsSessionContext::with_defaults();
        assert!(ctx.is_ok());
    }

    #[test]
    fn test_custom_config() {
        let config = MetricsQueryConfig::for_aggregation();
        let ctx = MetricsSessionContext::new(config);
        assert!(ctx.is_ok());
        assert_eq!(ctx.unwrap().config().max_memory_bytes, 1024 * 1024 * 1024);
    }
}
