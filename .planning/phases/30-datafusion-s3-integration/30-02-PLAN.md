    use std::sync::Arc;
    use quickwit_storage::Storage;
    use url::Url;
    use super::storage_bridge::StorageObjectStoreAdapter;

    impl MetricsSessionContext {
        /// Create a MetricsSessionContext with storage backend for S3/remote access.
        ///
        /// Registers the storage as an object store with DataFusion's RuntimeEnv,
        /// enabling queries against Parquet files on S3, Azure, GCS, etc.
        pub fn with_storage(
            config: MetricsQueryConfig,
            storage: Arc<dyn Storage>,
        ) -> Result<Self, ContextError> {
            // Create bridge adapter
            let adapter = Arc::new(StorageObjectStoreAdapter::new(storage.clone()));

            // Parse storage URI for registration
            let storage_uri = storage.uri();
            let url = Url::parse(storage_uri.as_str())
                .map_err(|e| ContextError::Config(format!("Invalid storage URI: {}", e)))?;

            // Build RuntimeEnv with object store
            let runtime_env = RuntimeEnvBuilder::new()
                .with_memory_limit(config.max_memory_bytes, 1.0)
                .with_object_store(&url, adapter)
                .build_arc()?;

            let session_config = SessionConfig::new()
                .with_target_partitions(config.target_partitions)
                .with_batch_size(config.batch_size)
                .with_parquet_pruning(config.parquet.prune_row_groups);

            let inner = SessionContext::new_with_config_rt(session_config, runtime_env);

            Ok(Self { inner, config })
        }
    }
    ```

    }

    pub fn with_storage_uri(
        schema: &MetricsSchema,
        splits: &[MetricsSplit],
        storage_uri: quickwit_common::uri::Uri,
    ) -> Self {
        let scheme = storage_uri.protocol().to_string();
        let bucket = storage_uri.filepath().and_then(|p| {
            // For S3 URIs like s3://bucket/prefix, extract bucket
            p.to_str()?.split('/').next().map(|s| s.to_string())
        });

        // Build full URIs for each parquet file
        let parquet_uris: Vec<String> = splits
            .iter()
            .flat_map(|split| {
                split.metadata.parquet_files.iter().map(|f| {
                    format!("{}/{}", storage_uri.as_str().trim_end_matches('/'), f)
                })
            })
            .collect();

        Self {
            schema: schema.arrow_schema().clone(),
            parquet_uris,
            storage_scheme: scheme,
            bucket,
        }
    }
    ```

    fn extract_path_from_uri(uri: &str, scheme: &str, bucket: &Option<String>) -> String {
        match scheme {
            "s3" => {
                // s3://bucket/path/to/file.parquet -> path/to/file.parquet
                if let Some(b) = bucket {
                    uri.strip_prefix(&format!("s3://{}/", b))
                        .unwrap_or(uri)
                        .to_string()
                } else {
                    uri.to_string()
                }
            }
            "file" => {
                // file:///path/to/file.parquet -> /path/to/file.parquet
                uri.strip_prefix("file://").unwrap_or(uri).to_string()
            }
            _ => uri.to_string(),
        }
    }
