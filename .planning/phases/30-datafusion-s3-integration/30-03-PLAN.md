    Update leaf_search_metrics_split to use storage-aware context instead of PathBuf conversion.

    Current broken code (lines 669-679):
    ```rust
    // Get storage base path for resolving parquet file paths
    // For file storage, use the filepath; for others, parse from URI string
    let base_path = storage
        .uri()
        .filepath()
        .map(PathBuf::from)
        .unwrap_or_else(|| PathBuf::from(storage.uri().as_str()));

    // Register the split as a table
    ctx.register_splits("metrics", &[metrics_split.clone()], &base_path)
    ```

    Replace with:
    ```rust
    // Create DataFusion context with storage support
    let config = MetricsQueryConfig::default();
    let ctx = MetricsSessionContext::with_storage(config, storage.clone())
        .map_err(|e| SearchError::Internal(format!("DataFusion context error: {}", e)))?;

    // Register the split using storage URI (works for S3 and local)
    ctx.register_splits_with_uri("metrics", &[metrics_split.clone()], storage.uri())
        .map_err(|e| SearchError::Internal(format!("Failed to register metrics table: {}", e)))?;
    ```

    This change:
    1. Uses with_storage() to register the storage adapter with DataFusion
    2. Uses register_splits_with_uri() with the actual storage URI
    3. Removes the PathBuf conversion that breaks for S3

    Add import at top of file:
    ```rust
    use quickwit_metrics_engine::query::MetricsSessionContext;
    ```

    If quickwit-search doesn't already depend on quickwit-metrics-engine, add to Cargo.toml:
    ```toml
    quickwit-metrics-engine = { workspace = true }
    ```

    Test structure:
    ```rust
    #[tokio::test]
    async fn test_metrics_query_via_object_storage() {
        // 1. Create test data
        let schema = MetricsSchema::new();
        let mut ingest_buffer = MetricsIngestBuffer::new(schema.clone(), Default::default());

        // Add test metrics
        ingest_buffer.add_metric(/* test data */);

        // 2. Write to Parquet in memory
        let record_batch = ingest_buffer.finish().unwrap();
        let parquet_bytes = write_parquet_to_bytes(&record_batch);

        // 3. Upload to storage (RamStorage simulates S3 API)
        let storage = Arc::new(RamStorage::default());
        let split_id = "test-split-001";
        let parquet_path = format!("{}/{}.parquet", split_id, split_id);
        storage.put(&Path::new(&parquet_path), parquet_bytes.into()).await.unwrap();

        // 4. Create split info
        let split = SplitIdAndFooterOffsets {
            split_id: split_id.to_string(),
            num_docs: record_batch.num_rows() as u64,
            timestamp_start: Some(1000),
            timestamp_end: Some(2000),
            ..Default::default()
        };

        // 5. Execute search
        let search_request = SearchRequest {
            max_hits: 10,
            ..Default::default()
        };

        let result = leaf_search_metrics_split(
            search_request,
            searcher_context,
            storage,
            split,
        ).await;

        // 6. Verify results
        assert!(result.is_ok());
        let response = result.unwrap();
        assert!(response.num_hits > 0);
        assert_eq!(response.num_successful_splits, 1);
    }
    ```

    If RamStorage doesn't work directly with DataFusion (since it's quickwit-storage not object_store), the test should still pass because StorageObjectStoreAdapter bridges the gap.
