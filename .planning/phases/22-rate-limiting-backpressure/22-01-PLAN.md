3. **Add convenience method for pre-append check:**
```rust
impl MetricsIngesterState {
    /// Checks both WAL capacity and rate limit before append.
    /// Call this before append_records to implement backpressure.
    pub async fn check_can_append(
        &self,
        queue_id: &str,
        bytes: ByteSize,
    ) -> Result<(), BackpressureError> {
        // Check WAL capacity first
        let wal = self.wal.read().await;
        wal.check_enough_capacity(bytes).await?;

        // Check rate limit
        let mut inner = self.inner.lock().await;
        if !inner.rate_limiter.acquire_bytes(queue_id, bytes) {
            let available = inner.rate_limiter.get_or_create(queue_id).available_permits();
            return Err(BackpressureError::RateLimitExceeded {
                queue_id: queue_id.to_string(),
                requested: bytes,
                available,
            });
        }
        Ok(())
    }
}
```
