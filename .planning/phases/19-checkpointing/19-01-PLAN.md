4. **Recovery on init pattern (from METRICS-DURABILITY-DESIGN.md Section 3.4):**
```rust
pub async fn init(&self) -> Result<(), ReadRecordError> {
    let wal = self.wal.write().await;
    let mut inner = self.inner.lock().await;

    for queue_id in wal.list_metrics_queues().await {
        if let Some(position_range) = wal.queue_position_range(&queue_id).await {
            // Non-empty queue: recover shard position
            let replication_position = Position::offset(*position_range.end());
            let truncation_position = if *position_range.start() == 0 {
                Position::Beginning
            } else {
                Position::offset(*position_range.start() - 1)
            };
            let shard_position = MetricsShardPosition::new(
                replication_position,
                truncation_position,
            );
            inner.shard_positions.insert(queue_id, shard_position);
        } else {
            // Empty queue: delete it
            wal.force_delete_queue(&queue_id).await?;
        }
    }
    Ok(())
}
```

```rust
pub struct FullyLockedMetricsState<'a> {
    pub wal: RwLockWriteGuard<'a, MetricsWal>,
    pub inner: MutexGuard<'a, MetricsIngesterInner>,
}

impl MetricsIngesterState {
    /// Acquires both locks in correct order (WAL first, then inner).
    pub async fn lock_fully(&self) -> FullyLockedMetricsState<'_> {
        let wal = self.wal.write().await;
        let inner = self.inner.lock().await;
        FullyLockedMetricsState { wal, inner }
    }
}
```

```rust
#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_shard_position_new_at_beginning() {
        let pos = MetricsShardPosition::new_at_beginning();
        assert_eq!(pos.replication_position_inclusive, Position::Beginning);
        assert_eq!(pos.truncation_position_inclusive, Position::Beginning);
    }

    #[test]
    fn test_update_replication_position_advances() {
        let mut pos = MetricsShardPosition::new_at_beginning();
        pos.update_replication_position(Position::offset(10));
        assert_eq!(pos.replication_position_inclusive, Position::offset(10));

        // Higher position updates
        pos.update_replication_position(Position::offset(20));
        assert_eq!(pos.replication_position_inclusive, Position::offset(20));

        // Lower position does NOT update
        pos.update_replication_position(Position::offset(15));
        assert_eq!(pos.replication_position_inclusive, Position::offset(20));
    }

    #[test]
    fn test_update_truncation_position_advances() {
        let mut pos = MetricsShardPosition::new(Position::offset(100), Position::Beginning);
        pos.update_truncation_position(Position::offset(50));
        assert_eq!(pos.truncation_position_inclusive, Position::offset(50));

        // Lower position does NOT update
        pos.update_truncation_position(Position::offset(25));
        assert_eq!(pos.truncation_position_inclusive, Position::offset(50));
    }

    #[test]
    fn test_can_truncate_to() {
        let pos = MetricsShardPosition::new(Position::offset(100), Position::offset(50));
        assert!(pos.can_truncate_to(Position::offset(50))); // exactly at truncation
        assert!(pos.can_truncate_to(Position::offset(25))); // before truncation
        assert!(!pos.can_truncate_to(Position::offset(75))); // after truncation
    }

    #[test]
    fn test_truncation_lag() {
        let pos = MetricsShardPosition::new(Position::offset(100), Position::offset(50));
        assert_eq!(pos.truncation_lag(), Some(50));

        let pos2 = MetricsShardPosition::new(Position::offset(100), Position::Beginning);
        assert_eq!(pos2.truncation_lag(), Some(101)); // 0..=100 is 101 records

        let pos3 = MetricsShardPosition::new_at_beginning();
        assert_eq!(pos3.truncation_lag(), None); // both at beginning, no lag
    }
}
```
