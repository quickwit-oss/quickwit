```rust
/// Result of a truncation operation.
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TruncationResult {
    /// Successfully truncated records up to the given position.
    Truncated {
        queue_id: String,
        position: u64,
        records_removed: usize,
    },
    /// No truncation needed (already truncated past this position).
    AlreadyTruncated { queue_id: String },
    /// Cannot truncate - would lose unindexed data.
    UnsafeTruncation {
        queue_id: String,
        requested_position: u64,
        safe_position: u64,
    },
    /// Queue does not exist (may have been deleted).
    QueueNotFound { queue_id: String },
}
```

2. **Safe truncation function:**
```rust
/// Safely truncate a queue up to a position, respecting the shard's truncation position.
///
/// This function ensures we never truncate records that haven't been indexed yet.
/// It checks the MetricsShardPosition's truncation_position_inclusive before truncating.
///
/// # Returns
/// - `TruncationResult::Truncated` - Successfully truncated
/// - `TruncationResult::AlreadyTruncated` - Position already past truncation point
/// - `TruncationResult::UnsafeTruncation` - Would lose unindexed data
/// - `TruncationResult::QueueNotFound` - Queue doesn't exist
pub async fn safe_truncate(
    wal: &MetricsWal,
    queue_id: &str,
    shard_position: &MetricsShardPosition,
    truncate_up_to: u64,
) -> Result<TruncationResult, io::Error> {
    // Check if queue exists
    if !wal.queue_exists(queue_id).await {
        return Ok(TruncationResult::QueueNotFound {
            queue_id: queue_id.to_string(),
        });
    }

    // Get the safe truncation position from shard state
    let safe_position = match &shard_position.truncation_position_inclusive {
        Position::Beginning => {
            // Nothing has been indexed yet, no safe truncation
            return Ok(TruncationResult::UnsafeTruncation {
                queue_id: queue_id.to_string(),
                requested_position: truncate_up_to,
                safe_position: 0,
            });
        }
        Position::Offset(offset) => *offset,
        Position::Eof(_) => {
            // All records indexed, can truncate anything
            u64::MAX
        }
    };

    // Check if requested truncation is safe
    if truncate_up_to > safe_position {
        return Ok(TruncationResult::UnsafeTruncation {
            queue_id: queue_id.to_string(),
            requested_position: truncate_up_to,
            safe_position,
        });
    }

    // Check current position range to see if already truncated
    if let Some(range) = wal.queue_position_range(queue_id).await {
        if *range.start() > truncate_up_to {
            return Ok(TruncationResult::AlreadyTruncated {
                queue_id: queue_id.to_string(),
            });
        }
    }

    // Perform the truncation
    let records_removed = wal.truncate(queue_id, truncate_up_to).await
        .map_err(|e| io::Error::new(io::ErrorKind::Other, e.to_string()))?;

    Ok(TruncationResult::Truncated {
        queue_id: queue_id.to_string(),
        position: truncate_up_to,
        records_removed,
    })
}
```

3. **Batch truncation helper:**
```rust
/// Truncate multiple queues based on checkpoint delta positions.
///
/// For each partition in the checkpoint delta, truncate the corresponding
/// WAL queue up to the delta's "to" position. Uses safe_truncate internally.
pub async fn truncate_from_checkpoint(
    state: &MetricsIngesterState,
    checkpoint_delta: &SourceCheckpointDelta,
    queue_id_prefix: &str,  // e.g., "metrics/index-id/source-id/"
) -> Vec<TruncationResult> {
    let mut results = Vec::new();

    let locked = state.lock_fully().await;

    for (partition_id, partition_delta) in checkpoint_delta.iter() {
        // Build queue ID from partition (shard ID)
        let queue_id = format!("{}{}", queue_id_prefix, partition_id);

        // Get the shard position if it exists
        let shard_position = match locked.inner.shard_positions.get(&queue_id) {
            Some(pos) => pos,
            None => {
                results.push(TruncationResult::QueueNotFound {
                    queue_id: queue_id.clone(),
                });
                continue;
            }
        };

        // Extract truncate position from delta
        let truncate_up_to = match &partition_delta.to {
            Position::Beginning => continue, // Nothing to truncate
            Position::Offset(offset) => *offset,
            Position::Eof(_) => continue, // Handle EOF case
        };

        match safe_truncate(&locked.wal, &queue_id, shard_position, truncate_up_to).await {
            Ok(result) => results.push(result),
            Err(e) => {
                tracing::error!(
                    queue_id = %queue_id,
                    error = %e,
                    "failed to truncate WAL queue"
                );
            }
        }
    }

    results
}
```

```rust
mod tests {
    use super::*;
    use crate::metrics::{MetricsWal, MetricsWalConfig, MetricsShardPosition};

    use bytesize::ByteSize;

    async fn create_test_wal() -> (MetricsWal, tempfile::TempDir) {
        let tempdir = tempdir().unwrap();
        let config = MetricsWalConfig {
            wal_dir_path: tempdir.path().to_path_buf(),
            disk_capacity: ByteSize::mb(256),
            memory_capacity: ByteSize::mb(64),
        };
        let wal = MetricsWal::open(config).await.unwrap();
        (wal, tempdir)
    }

    #[tokio::test]
    async fn test_safe_truncate_success() {
        let (wal, _tempdir) = create_test_wal().await;

        // Create queue and add records
        let queue_id = "metrics/test/source/shard-1";
        wal.create_queue(queue_id).await.unwrap();

        let records: Vec<_> = (0..10)
            .map(|i| MRecord::Doc(Bytes::from(format!("metric-{}", i))))
            .collect();
        let encoded: Vec<_> = records.iter().map(|r| r.encode()).collect();
        wal.append_records(queue_id, None, encoded.into_iter()).await.unwrap();

        // Set up shard position - truncation safe up to position 5
        let shard_position = MetricsShardPosition::new(
            Position::offset(9),  // replication at 9
            Position::offset(5),  // truncation safe to 5
        );

        // Truncate up to position 3 (should succeed)
        let result = safe_truncate(&wal, queue_id, &shard_position, 3).await.unwrap();

        assert!(matches!(
            result,
            TruncationResult::Truncated { records_removed: 4, .. } // 0,1,2,3
        ));

        // Verify remaining records start at 4
        let range = wal.queue_position_range(queue_id).await.unwrap();
        assert_eq!(*range.start(), 4);
    }
```
