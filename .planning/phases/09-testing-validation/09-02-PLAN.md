    fn build_gauge_metrics(metric_name: &str, value: f64, timestamp_nanos: u64) -> Vec<ResourceMetrics> {
        let data_point = NumberDataPoint {
            time_unix_nano: timestamp_nanos,
            as_double: value,
            ..Default::default()
        };

        let metric = Metric {
            name: metric_name.to_string(),
            data: Some(quickwit_proto::opentelemetry::proto::metrics::v1::metric::Data::Gauge(
                Gauge {
                    data_points: vec![data_point],
                }
            )),
            ..Default::default()
        };

        let scope_metrics = ScopeMetrics {
            metrics: vec![metric],
            ..Default::default()
        };

        let resource = Resource {
            attributes: vec![
                KeyValue {
                    key: "service.name".to_string(),
                    value: Some(AnyValue {
                        value: Some(Value::StringValue("test-service".to_string())),
                    }),
                },
            ],
            ..Default::default()
        };

        vec![ResourceMetrics {
            resource: Some(resource),
            scope_metrics: vec![scope_metrics],
            ..Default::default()
        }]
    }
    ```

    #[tokio::test]
    async fn test_ingest_metrics_with_otlp_grpc_api() {
        initialize_tests();

        let mut sandbox = ClusterSandboxBuilder::default()
            .add_node([QuickwitService::Searcher])
            .add_node([QuickwitService::Metastore])
            .add_node_with_otlp([QuickwitService::Indexer])
            .add_node([QuickwitService::ControlPlane])
            .add_node([QuickwitService::Janitor])
            .build_and_start()
            .await;

        // Wait for pipelines - logs, traces, and metrics
        sandbox.wait_for_indexing_pipelines(3).await.unwrap();

        let timestamp = 1724060143000000001u64; // Fixed timestamp for reproducibility

        // Test with plain and gzip-compressed clients
        let tested_clients = vec![
            sandbox.metrics_client().clone(),
            sandbox.metrics_client().clone().send_compressed(CompressionEncoding::Gzip),
        ];

        for (idx, mut tested_client) in tested_clients.into_iter().enumerate() {
            let metric_name = format!("test_gauge_{idx}");
            let request = ExportMetricsServiceRequest {
                resource_metrics: build_gauge_metrics(&metric_name, 42.0 + idx as f64, timestamp),
            };

            let response = tested_client.export(request).await.unwrap();
            let inner = response.into_inner();

            // Verify no metrics were rejected
            if let Some(partial_success) = inner.partial_success {
                assert_eq!(
                    partial_success.rejected_data_points, 0,
                    "Some data points were rejected"
                );
            }
        }

        sandbox
            .shutdown_services([QuickwitService::Indexer])
            .await
            .unwrap();
        sandbox.shutdown().await.unwrap();
    }
    ```

    Note: This test validates the OTLP ingest path works. It doesn't yet verify splits are created
    because MetricsSplits may not be registered with the metastore in the current implementation.
    The test focuses on verifying the ingest API accepts metrics without errors.
  </action>
