    ```rust
    // Copyright 2021-Present Datadog, Inc.

    //! End-to-end tests for metrics engine roundtrip correctness.

    use std::sync::Arc;
    use tempfile::TempDir;

    use arrow::array::{Float64Array, StringArray, TimestampNanosecondArray};
    use arrow::record_batch::RecordBatch;

    use crate::ingest::{MetricsBatchAccumulator, MetricsIngestConfig};
    use crate::query::{MetricsQueryContext, MetricsQueryConfig};
    use crate::schema::MetricsSchema;

    /// Test complete roundtrip: create batch → write split → query split → verify data
    #[tokio::test]
    async fn test_ingest_query_roundtrip() {
        let temp_dir = TempDir::new().unwrap();
        let schema = MetricsSchema::new();
        let arrow_schema = schema.arrow_schema().clone();

        // Create test data
        let metric_names = StringArray::from(vec!["cpu.usage", "cpu.usage", "memory.used"]);
        let values = Float64Array::from(vec![0.75, 0.80, 1024.0]);
        let timestamps = TimestampNanosecondArray::from(vec![
            1704067200_000_000_000i64, // 2024-01-01 00:00:00
            1704067260_000_000_000i64, // 2024-01-01 00:01:00
            1704067200_000_000_000i64, // 2024-01-01 00:00:00
        ]);
        let service_names = StringArray::from(vec!["web-server", "web-server", "web-server"]);

        let batch = RecordBatch::try_new(
            arrow_schema.clone(),
            vec![
                Arc::new(metric_names),
                Arc::new(values),
                Arc::new(timestamps),
                Arc::new(service_names),
            ],
        )
        .unwrap();

        // Ingest: accumulate batch and flush to split
        let config = MetricsIngestConfig::default();
        let mut accumulator = MetricsBatchAccumulator::new(config, temp_dir.path());

        accumulator.add_batch(batch).unwrap();
        let split = accumulator.flush().unwrap().expect("Should produce a split");

        // Verify split metadata
        assert_eq!(split.num_rows, 3);
        assert!(split.parquet_path.exists());

        // Query: load split and execute DataFusion query
        let query_config = MetricsQueryConfig::default();
        let query_ctx = MetricsQueryContext::new(query_config);

        // Register the split with DataFusion
        query_ctx
            .register_split("metrics", &split)
            .await
            .unwrap();

        // Query all rows
        let df = query_ctx
            .session()
            .sql("SELECT metric_name, value, timestamp FROM metrics ORDER BY timestamp")
            .await
            .unwrap();

        let results = df.collect().await.unwrap();
        assert_eq!(results.len(), 1);
        assert_eq!(results[0].num_rows(), 3);

        // Verify data integrity
        let metric_names = results[0]
            .column(0)
            .as_any()
            .downcast_ref::<StringArray>()
            .unwrap();
        assert_eq!(metric_names.value(0), "cpu.usage");
        assert_eq!(metric_names.value(1), "memory.used");
        assert_eq!(metric_names.value(2), "cpu.usage");

        let values = results[0]
            .column(1)
            .as_any()
            .downcast_ref::<Float64Array>()
            .unwrap();
        assert!((values.value(0) - 0.75).abs() < 0.001);
    }

    /// Test time range filtering works correctly
    #[tokio::test]
    async fn test_time_range_query() {
        let temp_dir = TempDir::new().unwrap();
        let schema = MetricsSchema::new();
        let arrow_schema = schema.arrow_schema().clone();

        // Create test data spanning multiple hours
        let metric_names = StringArray::from(vec!["cpu", "cpu", "cpu"]);
        let values = Float64Array::from(vec![1.0, 2.0, 3.0]);
        let timestamps = TimestampNanosecondArray::from(vec![
            1704067200_000_000_000i64, // Hour 0
            1704070800_000_000_000i64, // Hour 1
            1704074400_000_000_000i64, // Hour 2
        ]);
        let service_names = StringArray::from(vec!["svc", "svc", "svc"]);

        let batch = RecordBatch::try_new(
            arrow_schema,
            vec![
                Arc::new(metric_names),
                Arc::new(values),
                Arc::new(timestamps),
                Arc::new(service_names),
            ],
        )
        .unwrap();

        let config = MetricsIngestConfig::default();
        let mut accumulator = MetricsBatchAccumulator::new(config, temp_dir.path());
        accumulator.add_batch(batch).unwrap();
        let split = accumulator.flush().unwrap().unwrap();

        let query_ctx = MetricsQueryContext::new(MetricsQueryConfig::default());
        query_ctx.register_split("metrics", &split).await.unwrap();

        // Query with time range filter - should get only hour 1
        let df = query_ctx
            .session()
            .sql("SELECT value FROM metrics WHERE timestamp >= 1704070800000000000 AND timestamp < 1704074400000000000")
            .await
            .unwrap();

        let results = df.collect().await.unwrap();
        assert_eq!(results[0].num_rows(), 1);

        let values = results[0]
            .column(0)
            .as_any()
            .downcast_ref::<Float64Array>()
            .unwrap();
        assert!((values.value(0) - 2.0).abs() < 0.001);
    }
    ```

    Note: This test may need adjustment based on actual MetricsSchema field names and
