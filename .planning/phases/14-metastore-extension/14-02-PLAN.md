```rust
#[derive(Debug, Clone, Default, Serialize, Deserialize)]
pub struct ListMetricsSplitsQuery {
    /// Index ID to filter by (required).
    pub index_id: String,
    /// Split states to include.
    #[serde(default)]
    pub split_states: Vec<String>,
    /// Time range start (inclusive).
    pub time_range_start: Option<i64>,
    /// Time range end (inclusive).
    pub time_range_end: Option<i64>,
    /// Metric names to filter by (any match).
    #[serde(default)]
    pub metric_names: Vec<String>,
    /// Service tag filter.
    pub tag_service: Option<String>,
    /// Environment tag filter.
    pub tag_env: Option<String>,
    /// Datacenter tag filter.
    pub tag_datacenter: Option<String>,
    /// Region tag filter.
    pub tag_region: Option<String>,
    /// Host tag filter.
    pub tag_host: Option<String>,
    /// Limit number of results.
    pub limit: Option<usize>,
}

impl ListMetricsSplitsQuery {
    /// Create a query for all splits in an index.
    pub fn for_index(index_id: impl Into<String>) -> Self {
        Self {
            index_id: index_id.into(),
            split_states: vec!["Published".to_string()],
            ..Default::default()
        }
    }

        self.split_states = states;

        self
    }
```

```rust
/// Helper trait to build a [`StageMetricsSplitsRequest`] and deserialize its payload.
pub trait StageMetricsSplitsRequestExt {
    /// Creates a new [`StageMetricsSplitsRequest`] from a list of [`MetricsSplitMetadata`].
    fn try_from_splits_metadata(
        index_id: impl Into<String>,
        splits_metadata: impl IntoIterator<Item = MetricsSplitMetadata>,
    ) -> MetastoreResult<StageMetricsSplitsRequest>;

    /// Deserializes the `split_metadata_list_serialized_json` field.
    fn deserialize_splits_metadata(&self) -> MetastoreResult<Vec<MetricsSplitMetadata>>;
}

impl StageMetricsSplitsRequestExt for StageMetricsSplitsRequest {
    fn try_from_splits_metadata(
        index_id: impl Into<String>,
        splits_metadata: impl IntoIterator<Item = MetricsSplitMetadata>,
    ) -> MetastoreResult<StageMetricsSplitsRequest> {
        let splits_metadata: Vec<MetricsSplitMetadata> = splits_metadata.into_iter().collect();
        let split_metadata_list_serialized_json = serde_utils::to_json_str(&splits_metadata)?;
        let request = Self {
            index_id: index_id.into(),
            split_metadata_list_serialized_json,
        };
        Ok(request)
    }

    fn deserialize_splits_metadata(&self) -> MetastoreResult<Vec<MetricsSplitMetadata>> {
        serde_utils::from_json_str(&self.split_metadata_list_serialized_json)
    }
}

/// Helper trait to build a [`ListMetricsSplitsRequest`] and deserialize its payload.
pub trait ListMetricsSplitsRequestExt {
    /// Creates a new [`ListMetricsSplitsRequest`] from a [`ListMetricsSplitsQuery`].
    fn try_from_query(query: &ListMetricsSplitsQuery) -> MetastoreResult<ListMetricsSplitsRequest>;

    /// Deserializes the `query_json` field.
    fn deserialize_query(&self) -> MetastoreResult<ListMetricsSplitsQuery>;
}

impl ListMetricsSplitsRequestExt for ListMetricsSplitsRequest {
    fn try_from_query(query: &ListMetricsSplitsQuery) -> MetastoreResult<ListMetricsSplitsRequest> {
        let query_json = serde_utils::to_json_str(query)?;
        Ok(Self { index_id: query.index_id.clone(), query_json })
    }

    fn deserialize_query(&self) -> MetastoreResult<ListMetricsSplitsQuery> {
        serde_utils::from_json_str(&self.query_json)
    }
}

/// Helper trait for [`ListMetricsSplitsResponse`].
pub trait ListMetricsSplitsResponseExt {
    /// Creates a new [`ListMetricsSplitsResponse`] from a list of [`MetricsSplit`].
    fn try_from_splits(splits: impl IntoIterator<Item = MetricsSplit>) -> MetastoreResult<ListMetricsSplitsResponse>;

    /// Deserializes the `splits_serialized_json` field.
    fn deserialize_splits(&self) -> MetastoreResult<Vec<MetricsSplit>>;

    /// Creates an empty response.
    fn empty() -> Self;
}

impl ListMetricsSplitsResponseExt for ListMetricsSplitsResponse {
    fn try_from_splits(splits: impl IntoIterator<Item = MetricsSplit>) -> MetastoreResult<ListMetricsSplitsResponse> {
        let splits: Vec<MetricsSplit> = splits.into_iter().collect();
        let splits_serialized_json = serde_utils::to_json_str(&splits)?;
        Ok(Self { splits_serialized_json })
    }

    fn deserialize_splits(&self) -> MetastoreResult<Vec<MetricsSplit>> {
        serde_utils::from_json_str(&self.splits_serialized_json)
    }

    fn empty() -> Self {
        Self {
            splits_serialized_json: "[]".to_string(),
        }
    }
}
```

```rust
    async fn stage_metrics_splits(
        &self,
        request: StageMetricsSplitsRequest,
    ) -> MetastoreResult<EmptyResponse> {
        let splits_metadata: Vec<MetricsSplitMetadata> =
            request.deserialize_splits_metadata()?;

        if splits_metadata.is_empty() {
            return Ok(EmptyResponse {});
        }

        // Prepare batch arrays for UNNEST
        let mut split_ids: Vec<String> = Vec::with_capacity(splits_metadata.len());
        let mut index_ids: Vec<String> = Vec::with_capacity(splits_metadata.len());
        let mut time_range_starts: Vec<i64> = Vec::with_capacity(splits_metadata.len());
        let mut time_range_ends: Vec<i64> = Vec::with_capacity(splits_metadata.len());
        let mut metric_names_list: Vec<Vec<String>> = Vec::with_capacity(splits_metadata.len());
        let mut tag_service_list: Vec<Option<Vec<String>>> = Vec::with_capacity(splits_metadata.len());
        let mut tag_env_list: Vec<Option<Vec<String>>> = Vec::with_capacity(splits_metadata.len());
        let mut tag_datacenter_list: Vec<Option<Vec<String>>> = Vec::with_capacity(splits_metadata.len());
        let mut tag_region_list: Vec<Option<Vec<String>>> = Vec::with_capacity(splits_metadata.len());
        let mut tag_host_list: Vec<Option<Vec<String>>> = Vec::with_capacity(splits_metadata.len());
        let mut high_cardinality_keys_list: Vec<Vec<String>> = Vec::with_capacity(splits_metadata.len());
        let mut num_rows_list: Vec<i64> = Vec::with_capacity(splits_metadata.len());
        let mut size_bytes_list: Vec<i64> = Vec::with_capacity(splits_metadata.len());
        let mut metadata_json_list: Vec<String> = Vec::with_capacity(splits_metadata.len());

        for metadata in &splits_metadata {
            let insertable = InsertableMetricsSplit::from_metadata(
                metadata,
                MetricsSplitState::Staged,
            ).map_err(|e| MetastoreError::Internal {
                message: format!("failed to serialize metrics split: {}", e),
                cause: e.to_string(),
            })?;

            split_ids.push(insertable.split_id);
            index_ids.push(insertable.index_id);
            time_range_starts.push(insertable.time_range_start);
            time_range_ends.push(insertable.time_range_end);
            metric_names_list.push(insertable.metric_names);
            tag_service_list.push(insertable.tag_service);
            tag_env_list.push(insertable.tag_env);
            tag_datacenter_list.push(insertable.tag_datacenter);
            tag_region_list.push(insertable.tag_region);
            tag_host_list.push(insertable.tag_host);
            high_cardinality_keys_list.push(insertable.high_cardinality_tag_keys);
            num_rows_list.push(insertable.num_rows);
            size_bytes_list.push(insertable.size_bytes);
            metadata_json_list.push(insertable.split_metadata_json);
        }

        run_with_tx!(self.connection_pool, tx, "stage metrics splits", {
            let upserted_split_ids: Vec<String> = sqlx::query_scalar(r#"
                INSERT INTO metrics_splits
                    (split_id, split_state, index_id, time_range_start, time_range_end,
                     metric_names, tag_service, tag_env, tag_datacenter, tag_region, tag_host,
                     high_cardinality_tag_keys, num_rows, size_bytes, split_metadata_json)
                SELECT
                    unnest($1::text[]),
                    'Staged',
                    unnest($2::text[]),
                    unnest($3::bigint[]),
                    unnest($4::bigint[]),
                    unnest($5::text[][]),
                    unnest($6::text[][]),
                    unnest($7::text[][]),
                    unnest($8::text[][]),
                    unnest($9::text[][]),
                    unnest($10::text[][]),
                    unnest($11::text[][]),
                    unnest($12::bigint[]),
                    unnest($13::bigint[]),
                    unnest($14::text[])
                ON CONFLICT(split_id) DO UPDATE
                    SET split_state = excluded.split_state,
                        time_range_start = excluded.time_range_start,
                        time_range_end = excluded.time_range_end,
                        metric_names = excluded.metric_names,
                        tag_service = excluded.tag_service,
                        tag_env = excluded.tag_env,
                        tag_datacenter = excluded.tag_datacenter,
                        tag_region = excluded.tag_region,
                        tag_host = excluded.tag_host,
                        high_cardinality_tag_keys = excluded.high_cardinality_tag_keys,
                        num_rows = excluded.num_rows,
                        size_bytes = excluded.size_bytes,
                        split_metadata_json = excluded.split_metadata_json,
                        update_timestamp = (CURRENT_TIMESTAMP AT TIME ZONE 'UTC')
                    WHERE metrics_splits.split_state = 'Staged'
                RETURNING split_id
            "#)
            .bind(&split_ids)
            .bind(&index_ids)
            .bind(&time_range_starts)
            .bind(&time_range_ends)
            .bind(&metric_names_list)
            .bind(&tag_service_list)
            .bind(&tag_env_list)
            .bind(&tag_datacenter_list)
            .bind(&tag_region_list)
            .bind(&tag_host_list)
            .bind(&high_cardinality_keys_list)
            .bind(&num_rows_list)
            .bind(&size_bytes_list)
            .bind(&metadata_json_list)
            .fetch_all(tx.as_mut())
            .await?;

            if upserted_split_ids.len() != split_ids.len() {
                let failed_splits: Vec<String> = split_ids
                    .into_iter()
                    .filter(|id| !upserted_split_ids.contains(id))
                    .collect();
                return Err(MetastoreError::FailedPrecondition {
                    entity: EntityKind::Splits { split_ids: failed_splits },
                    message: "metrics splits are not in staged state".to_string(),
                });
            }
            Ok(EmptyResponse {})
        })
    }
```

```rust
    async fn publish_metrics_splits(
        &self,
        request: PublishMetricsSplitsRequest,
    ) -> MetastoreResult<EmptyResponse> {
        if request.staged_split_ids.is_empty() && request.replaced_split_ids.is_empty() {
            return Ok(EmptyResponse {});
        }

        run_with_tx!(self.connection_pool, tx, "publish metrics splits", {
            // Publish staged splits
            if !request.staged_split_ids.is_empty() {
                let published_count = sqlx::query(r#"
                    UPDATE metrics_splits
                    SET split_state = 'Published',
                        update_timestamp = (CURRENT_TIMESTAMP AT TIME ZONE 'UTC')
                    WHERE split_id = ANY($1)
                      AND split_state = 'Staged'
                      AND index_id = $2
                "#)
                .bind(&request.staged_split_ids)
                .bind(&request.index_id)
                .execute(tx.as_mut())
                .await?
                .rows_affected();

                if published_count != request.staged_split_ids.len() as u64 {
                    return Err(MetastoreError::FailedPrecondition {
                        entity: EntityKind::Splits { split_ids: request.staged_split_ids.clone() },
                        message: "some metrics splits are not in staged state".to_string(),
                    });
                }
            }

            // Mark replaced splits for deletion (for merge operations)
            if !request.replaced_split_ids.is_empty() {
                sqlx::query(r#"
                    UPDATE metrics_splits
                    SET split_state = 'MarkedForDeletion',
                        update_timestamp = (CURRENT_TIMESTAMP AT TIME ZONE 'UTC')
                    WHERE split_id = ANY($1)
                      AND split_state = 'Published'
                      AND index_id = $2
                "#)
                .bind(&request.replaced_split_ids)
                .bind(&request.index_id)
                .execute(tx.as_mut())
                .await?;
            }

            Ok(EmptyResponse {})
        })
    }

    async fn list_metrics_splits(
        &self,
        request: ListMetricsSplitsRequest,
    ) -> MetastoreResult<ListMetricsSplitsResponse> {
        let query: ListMetricsSplitsQuery = request.deserialize_query()?;

        // Build dynamic query with filters
        let rows: Vec<PgMetricsSplit> = sqlx::query_as(r#"
            SELECT
                split_id, split_state, index_id, time_range_start, time_range_end,
                metric_names, tag_service, tag_env, tag_datacenter, tag_region, tag_host,
                high_cardinality_tag_keys, num_rows, size_bytes, split_metadata_json
            FROM metrics_splits
            WHERE index_id = $1
              AND ($2::text[] IS NULL OR split_state = ANY($2))
              AND ($3::bigint IS NULL OR time_range_start <= $3)
              AND ($4::bigint IS NULL OR time_range_end >= $4)
              AND ($5::text[] IS NULL OR metric_names && $5)
              AND ($6::text IS NULL OR $6 = ANY(tag_service))
              AND ($7::text IS NULL OR $7 = ANY(tag_env))
              AND ($8::text IS NULL OR $8 = ANY(tag_datacenter))
              AND ($9::text IS NULL OR $9 = ANY(tag_region))
              AND ($10::text IS NULL OR $10 = ANY(tag_host))
            ORDER BY time_range_start DESC
            LIMIT $11
        "#)
        .bind(&query.index_id)
        .bind(if query.split_states.is_empty() { None } else { Some(&query.split_states) })
        .bind(query.time_range_end)  // Note: inverted for overlap check
        .bind(query.time_range_start)
        .bind(if query.metric_names.is_empty() { None } else { Some(&query.metric_names) })
        .bind(&query.tag_service)
        .bind(&query.tag_env)
        .bind(&query.tag_datacenter)
        .bind(&query.tag_region)
        .bind(&query.tag_host)
        .bind(query.limit.map(|l| l as i64).unwrap_or(10000))
        .fetch_all(&self.connection_pool)
        .await?;

        let splits: Vec<MetricsSplit> = rows
            .into_iter()
            .filter_map(|row| MetricsSplit::try_from(row).ok())
            .collect();

        ListMetricsSplitsResponseExt::try_from_splits(splits)
    }

    async fn mark_metrics_splits_for_deletion(
        &self,
        request: MarkMetricsSplitsForDeletionRequest,
    ) -> MetastoreResult<EmptyResponse> {
        if request.split_ids.is_empty() {
            return Ok(EmptyResponse {});
        }

        sqlx::query(r#"
            UPDATE metrics_splits
            SET split_state = 'MarkedForDeletion',
                update_timestamp = (CURRENT_TIMESTAMP AT TIME ZONE 'UTC')
            WHERE split_id = ANY($1)
              AND index_id = $2
              AND split_state IN ('Staged', 'Published')
        "#)
        .bind(&request.split_ids)
        .bind(&request.index_id)
        .execute(&self.connection_pool)
        .await?;

        Ok(EmptyResponse {})
    }

    async fn delete_metrics_splits(
        &self,
        request: DeleteMetricsSplitsRequest,
    ) -> MetastoreResult<EmptyResponse> {
        if request.split_ids.is_empty() {
            return Ok(EmptyResponse {});
        }

        let deleted_count = sqlx::query(r#"
            DELETE FROM metrics_splits
            WHERE split_id = ANY($1)
              AND index_id = $2
              AND split_state = 'MarkedForDeletion'
        "#)
        .bind(&request.split_ids)
        .bind(&request.index_id)
        .execute(&self.connection_pool)
        .await?
        .rows_affected();

        if deleted_count != request.split_ids.len() as u64 {
            return Err(MetastoreError::FailedPrecondition {
                entity: EntityKind::Splits { split_ids: request.split_ids.clone() },
                message: "some metrics splits are not marked for deletion".to_string(),
            });
        }

        Ok(EmptyResponse {})
    }
```
