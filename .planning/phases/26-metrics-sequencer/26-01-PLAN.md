```rust
    pub async fn get_sender(
        &self,

    ) -> anyhow::Result<MetricsSplitsUpdateSender> {
        match self {
```

```rust
#[tokio::test]
async fn test_metrics_uploader_with_sequencer_ordering() {
    quickwit_common::setup_logging_for_tests();

    let universe = Universe::new();

    // Create publisher and sequencer chain
    let (publisher_mailbox, publisher_inbox) = universe.create_test_mailbox::<MetricsPublisher>();
    let sequencer = Sequencer::new(publisher_mailbox);
    let (sequencer_mailbox, _sequencer_handle) = universe.spawn_builder().spawn(sequencer);

    let mut mock_metastore = MockMetastoreService::new();
    // Allow multiple stage calls
    mock_metastore
        .expect_stage_metrics_splits()
        .returning(|_| Ok(EmptyResponse {}));

    let ram_storage = Arc::new(RamStorage::default());
    let uploader = MetricsUploader::new(
        UploaderType::IndexUploader,
        MetastoreServiceClient::from_mock(mock_metastore),
        ram_storage.clone(),
        MetricsSplitsUpdateMailbox::Sequencer(sequencer_mailbox),
        4,
    );

    let (uploader_mailbox, uploader_handle) = universe.spawn_builder().spawn(uploader);

    // Send batches in order: split-1, split-2, split-3
    for i in 1..=3 {
        let splits = vec![create_test_metrics_split("test-index", &format!("split-{}", i))];
        let checkpoint_delta = IndexCheckpointDelta {
            source_id: "test-source".to_string(),
            source_delta: SourceCheckpointDelta::from_range((i*10)..(i*10+10)),
        };
        let batch = MetricsSplitBatch::new(
            splits,
            checkpoint_delta,
            PublishLock::default(),
            None,
        );
        uploader_mailbox.send_message(batch).await.unwrap();
    }

    // Wait for all messages to be processed
    uploader_handle.process_pending_and_observe().await;

    // Collect received messages from publisher inbox
    let mut received_ids = Vec::new();
    while let Ok(msg) = publisher_inbox.try_recv() {
        if let Some(update) = msg.downcast_ref::<MetricsSplitsUpdate>() {
            for split in &update.new_splits {
                received_ids.push(split.split_id.as_str().to_string());
            }
        }
    }

    // Verify ordering: should receive in order split-1, split-2, split-3
    // (Sequencer ensures FIFO delivery even if uploads complete out of order)
    assert_eq!(received_ids, vec!["split-1", "split-2", "split-3"]);

    universe.assert_quit().await;
}
```
