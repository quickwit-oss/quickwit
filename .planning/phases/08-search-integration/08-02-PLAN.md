Add function to load MetricsSplit metadata from storage.

MetricsSplits are stored as JSON metadata alongside Parquet data files.
The split_id maps to a `.split` file that contains the serialized MetricsSplit.

For metrics splits, the storage layout is:
- `{split_id}.split` - JSON metadata (MetricsSplit serialized)
- Parquet files referenced in metadata.parquet_files

Add to metrics_leaf.rs:

```rust
use std::path::PathBuf;
use std::time::Instant;

use quickwit_metrics_engine::split::MetricsSplit;
use quickwit_metrics_engine::query::{MetricsSessionContext, MetricsQueryConfig};
use quickwit_metrics_engine::schema::MetricsSchema;

/// Load MetricsSplit metadata from storage.
///
/// Reads the split metadata JSON file and deserializes it.
async fn load_metrics_split(
    storage: &dyn Storage,
    split_id: &str,
) -> Result<MetricsSplit, SearchError> {
    let split_file = PathBuf::from(format!("{}.split", split_id));

    let bytes = storage
        .get_all(&split_file)
        .await
        .map_err(|e| SearchError::Internal(format!(
            "failed to load metrics split '{}': {}",
            split_id, e
        )))?;

    MetricsSplit::from_json(&bytes)
        .map_err(|e| SearchError::Internal(format!(
            "failed to parse metrics split '{}': {}",
            split_id, e
        )))
}
```

Note: This assumes metrics splits are stored as JSON. If they use a different format
(e.g., postcard serialization like Tantivy splits), adjust the deserialization.
Check how MetricsSplit is written in the indexing pipeline to match the format.

use arrow::array::{Array, Float64Array, StringArray, UInt64Array};
use arrow::record_batch::RecordBatch;
use datafusion::prelude::*;

/// Execute a leaf search on a metrics split using DataFusion.
pub async fn leaf_search_metrics_split(
    search_request: SearchRequest,
    searcher_context: Arc<SearcherContext>,
    storage: Arc<dyn Storage>,
    split: SplitIdAndFooterOffsets,
) -> crate::Result<LeafSearchResponse> {
    let start_time = Instant::now();
    let split_id = split.split_id.clone();

    // Load the metrics split metadata
    let metrics_split = load_metrics_split(storage.as_ref(), &split_id).await?;

    // Create DataFusion context
    let config = MetricsQueryConfig::default();
    let ctx = MetricsSessionContext::new(config)
        .map_err(|e| SearchError::Internal(format!("DataFusion context error: {}", e)))?;

    // Get storage base path for resolving parquet file paths
    // The storage URI gives us the base path
    let base_path = PathBuf::from(storage.uri().path());

    // Register the split as a table
    ctx.register_splits("metrics", &[metrics_split.clone()], &base_path)
        .map_err(|e| SearchError::Internal(format!("Failed to register metrics table: {}", e)))?;

    // Convert query AST to SQL
    // For now, execute a simple SELECT * with any filters from the request
    // Future: Properly translate QueryAst to SQL predicates
    let sql = build_metrics_sql(&search_request)?;

    // Execute query
    let df = ctx.sql(&sql).await
        .map_err(|e| SearchError::Internal(format!("DataFusion query error: {}", e)))?;

    // Collect results
    let batches = df.collect().await
        .map_err(|e| SearchError::Internal(format!("DataFusion collect error: {}", e)))?;

    // Convert to LeafSearchResponse
    let (num_hits, partial_hits) = convert_batches_to_hits(&batches, &split_id, &search_request)?;

    let cpu_time = start_time.elapsed();

    Ok(LeafSearchResponse {
        num_hits,
        partial_hits,
        failed_splits: Vec::new(),
        num_attempted_splits: 1,
        num_successful_splits: 1,
        intermediate_aggregation_result: None, // TODO: Handle aggregations in future
        resource_stats: Some(ResourceStats {
            cpu_microsecs: cpu_time.as_micros() as u64,
            short_lived_cache_num_bytes: 0,
            split_num_docs: split.num_docs,
            warmup_microsecs: 0,
            cpu_thread_pool_wait_microsecs: 0,
        }),
    })
}

/// Build SQL query from SearchRequest.
///
/// Translates the search request into a SQL query for DataFusion.
/// For MVP: Simple SELECT with LIMIT. Future: Full QueryAst translation.
fn build_metrics_sql(request: &SearchRequest) -> Result<String, SearchError> {
    // Start with basic SELECT
    let mut sql = "SELECT * FROM metrics".to_string();

    // Add time range filter if present
    if let (Some(start), Some(end)) = (request.start_timestamp, request.end_timestamp) {
        sql.push_str(&format!(
            " WHERE timestamp_secs >= {} AND timestamp_secs < {}",
            start, end
        ));
    }

    // Add LIMIT if max_hits specified
    if request.max_hits > 0 {
        sql.push_str(&format!(" LIMIT {}", request.max_hits));
    }

    Ok(sql)
}

/// Convert DataFusion RecordBatches to PartialHits.
///
/// Each row in the result becomes a PartialHit.
fn convert_batches_to_hits(
    batches: &[RecordBatch],
    split_id: &str,
    request: &SearchRequest,
) -> Result<(u64, Vec<PartialHit>), SearchError> {
    let mut partial_hits = Vec::new();
    let mut total_rows = 0u64;

    for batch in batches {
        let num_rows = batch.num_rows();
        total_rows += num_rows as u64;

        for row_idx in 0..num_rows {
            // Create a PartialHit for each row
            // doc_id is the row index within this batch
            // segment_ord is 0 since Parquet doesn't have segments
            let hit = PartialHit {
                split_id: split_id.to_string(),
                segment_ord: 0,
                doc_id: row_idx as u32,
                sort_value: extract_sort_value(batch, row_idx, request),
                sort_value2: None,
            };
            partial_hits.push(hit);
        }
    }

    Ok((total_rows, partial_hits))
}

/// Extract sort value from a row if sorting is requested.
fn extract_sort_value(
    batch: &RecordBatch,
    row_idx: usize,
    request: &SearchRequest,
) -> Option<quickwit_proto::search::SortByValue> {
    // If no sort fields specified, return None
    if request.sort_fields.is_empty() {
        return None;
    }

    // Try to get the first sort field value
    let sort_field = &request.sort_fields[0];
    let field_name = &sort_field.field_name;

    // Find column by name
    let schema = batch.schema();
    let col_idx = schema.index_of(field_name).ok()?;
    let column = batch.column(col_idx);

    // Extract value based on column type
    if let Some(arr) = column.as_any().downcast_ref::<UInt64Array>() {
        if !arr.is_null(row_idx) {
            return Some(quickwit_proto::search::SortByValue {
                sort_value: Some(quickwit_proto::search::sort_by_value::SortValue::U64(
                    arr.value(row_idx)
                )),
            });
        }
    } else if let Some(arr) = column.as_any().downcast_ref::<Float64Array>() {
        if !arr.is_null(row_idx) {
            return Some(quickwit_proto::search::SortByValue {
                sort_value: Some(quickwit_proto::search::sort_by_value::SortValue::F64(
                    arr.value(row_idx)
                )),
            });
        }
    }

    None
}
