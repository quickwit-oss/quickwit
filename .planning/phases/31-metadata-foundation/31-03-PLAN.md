```rust
async fn list_metrics_splits_for_compaction(
    &mut self,
    request: ListMetricsSplitsForCompactionRequest,
) -> MetastoreResult<ListMetricsSplitsResponse> {
    // Query Published splits matching (index_id, window_start, sort_schema).
    // Uses the composite index idx_metrics_splits_compaction_scope.
    let query = r#"
        SELECT split_id, split_state, index_id, time_range_start, time_range_end,
               metric_names, tag_service, tag_env, tag_datacenter, tag_region, tag_host,
               high_cardinality_tag_keys, num_rows, size_bytes, split_metadata_json,
               window_start, window_duration_secs, sort_schema, num_merge_ops,
               row_keys, zonemap_regexes
        FROM metrics_splits
        WHERE index_id = $1
          AND window_start = $2
          AND sort_schema = $3
          AND split_state = 'Published'
        ORDER BY time_range_start
    "#;

    let rows = sqlx::query_as(query)
        .bind(&request.index_id)
        .bind(request.window_start)
        .bind(&request.sort_schema)
        .fetch_all(&self.connection_pool)
        .await
        .map_err(|err| {
            MetastoreError::Internal {
                message: format!("failed to list metrics splits for compaction: {err}"),
                cause: err.to_string(),
            }
        })?;

    // Convert PgMetricsSplit rows to MetricsSplitRecord and serialize as JSON
    let split_records: Vec<MetricsSplitRecord> = rows
        .into_iter()
        .filter_map(|row: PgMetricsSplit| MetricsSplitRecord::try_from(row).ok())
        .collect();

    let splits_json = serde_json::to_string(&split_records)
        .map_err(|err| MetastoreError::Internal {
            message: "failed to serialize splits".to_string(),
            cause: err.to_string(),
        })?;

    Ok(ListMetricsSplitsResponse {
        splits_serialized_json: splits_json,
    })
}
```

Update the implementation to execute both operations in a single PostgreSQL transaction:

```rust
async fn publish_metrics_splits(
    &mut self,
    request: PublishMetricsSplitsRequest,
) -> MetastoreResult<EmptyResponse> {
    let mut tx = self.connection_pool.begin().await.map_err(|err| {
        MetastoreError::Internal {
            message: "failed to begin transaction".to_string(),
            cause: err.to_string(),
        }
    })?;

    // Step 1: Publish staged splits (Staged -> Published)
    if !request.staged_split_ids.is_empty() {
        let publish_query = r#"
            UPDATE metrics_splits
            SET split_state = 'Published',
                update_timestamp = (CURRENT_TIMESTAMP AT TIME ZONE 'UTC')
            WHERE split_id = ANY($1)
              AND index_id = $2
              AND split_state = 'Staged'
        "#;

        let result = sqlx::query(publish_query)
            .bind(&request.staged_split_ids)
            .bind(&request.index_id)
            .execute(&mut *tx)
            .await
            .map_err(|err| MetastoreError::Internal {
                message: format!("failed to publish metrics splits: {err}"),
                cause: err.to_string(),
            })?;

        let published_count = result.rows_affected();
        let expected_count = request.staged_split_ids.len() as u64;
        if published_count != expected_count {
            // Rollback happens automatically when tx is dropped
            return Err(MetastoreError::Internal {
                message: format!(
                    "expected to publish {} splits, but only {} were in Staged state",
                    expected_count, published_count
                ),
                cause: "split state mismatch".to_string(),
            });
        }
    }

    // Step 2: Mark replaced splits for deletion (Published -> MarkedForDeletion)
    if !request.replaced_split_ids.is_empty() {
        let replace_query = r#"
            UPDATE metrics_splits
            SET split_state = 'MarkedForDeletion',
                update_timestamp = (CURRENT_TIMESTAMP AT TIME ZONE 'UTC')
            WHERE split_id = ANY($1)
              AND index_id = $2
              AND split_state = 'Published'
        "#;

        let result = sqlx::query(replace_query)
            .bind(&request.replaced_split_ids)
            .bind(&request.index_id)
            .execute(&mut *tx)
            .await
            .map_err(|err| MetastoreError::Internal {
                message: format!("failed to mark replaced metrics splits: {err}"),
                cause: err.to_string(),
            })?;

        let replaced_count = result.rows_affected();
        let expected_count = request.replaced_split_ids.len() as u64;
        if replaced_count != expected_count {
            return Err(MetastoreError::Internal {
                message: format!(
                    "expected to replace {} splits, but only {} were in Published state",
                    expected_count, replaced_count
                ),
                cause: "split state mismatch".to_string(),
            });
        }
    }

    // Both operations succeeded -- commit atomically
    tx.commit().await.map_err(|err| MetastoreError::Internal {
        message: "failed to commit publish transaction".to_string(),
        cause: err.to_string(),
    })?;

    Ok(EmptyResponse {})
}
```

**Key invariants:**
- Both staged_split_ids publish AND replaced_split_ids deletion happen in ONE transaction
- If any split is not in the expected state, the transaction rolls back entirely
- Row count verification ensures we don't silently skip splits that were already published/deleted by a concurrent operation

**2. Handle edge cases:**
- Empty `replaced_split_ids` (normal non-merge publish): Skip the replace step, just publish
- Empty `staged_split_ids` with non-empty `replaced_split_ids`: Skip publish step (edge case, may not occur in practice but should be handled)
- Both empty: No-op, return success
